```js

  if(w==='役職と 所属')      return can('post') || can('master');
  if(w==='レッスンの 日程')  return can('sched_all') || can('sched_mine');
  if(w==='レッスンの 出席')  return can('shukketsu') && (can('sched_all') || can('meibo'));

function expList(){ return Object.keys(EXP_COLS).filter(expCan) }

★**★決まりは 2つです。**

```
